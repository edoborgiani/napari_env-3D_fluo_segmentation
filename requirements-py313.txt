# Python 3.13 variant of requirements.txt, for use with
# Fluo_3D_nuc_seg_v1.6_py313.ipynb. Everything below is identical to
# requirements.txt except the two pins called out here:
#
# - numpy: 1.26.4 has no Python 3.13 wheels (numpy only ships cp313 wheels
#   from 2.1.0 onward), so it is bumped to numpy>=2.1. Its floor of <2.0 in
#   requirements.txt existed solely because tensorflow<2.17 hard-requires
#   numpy<2.0 -- moot here since tensorflow itself has to move for 3.13 (see
#   below), and scipy==1.14.1 / opencv-python==4.10.0.84 / aicsimageio==4.14.0
#   all already accept numpy>=2 with no upper bound.
# - tensorflow: 2.16.x only ships wheels up to Python 3.12; the first release
#   with cp313 wheels is 2.20.0, so the pin moves to tensorflow>=2.20.
#
# Everything else in this file (scipy, opencv-python, tetgen, meshlib,
# simpleitk, etc.) already publishes Python 3.13 wheels at the
# versions/floors requirements.txt pins, so no other changes were needed --
# EXCEPT one transitive pin: aicsimageio==4.14.0 unconditionally requires
# lxml<5, and lxml<5 has no Python 3.13 wheels (it fails to build from
# source on Windows without a full MSVC toolchain). Install this file with
# the companion overrides-py313.txt to fix that -- see that file for why
# the override is safe:
#   uv pip install -r requirements-py313.txt --override overrides-py313.txt
# (plain pip has no equivalent override mechanism -- this requires uv)
#
# This variant has not been run end-to-end -- please report issues.

# Core / visualization

# cuts new releases very often, so leaving it unpinned makes pip's resolver
# backtrack through hundreds of dask versions. Pinning skips that search.
dask[array]==2026.7.1
# Bumped for Python 3.13 wheel availability -- see header comment above.
numpy>=2.1
scipy==1.14.1

# with the dask pin above and sent pip walking back through aicsimageio's
# whole history (down to versions that predate the 'nd2' extra) looking for
# an alternative. 4.14.0 has no dask ceiling and includes the nd2 extra.

nd2reader
