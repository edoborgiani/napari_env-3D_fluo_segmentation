# Works for Python 3.10-3.13 (3.10/3.11 most tested; 3.12 and 3.13 verified
# against current PyPI wheel availability but less exercised in practice):
# numpy/tensorflow/lxml below are pinned per Python version via markers,
# since their <3.13 pins have no Python 3.13 wheels. Install with (uv
# required -- see the lxml note below for why):
#   uv pip install -r requirements.txt --override overrides.txt
#
# Core / visualization
napari[all]
# Pinned explicitly (not just via napari[all]'s "pyqt6" extra): napari can
# resolve without a Qt binding actually getting installed in some resolver/
# override combinations, which fails at runtime ("No Qt bindings could be
# found") rather than at install time. Requiring it directly here guarantees
# one is present regardless of how napari's extras resolve.
PyQt6
# Pinned: napari/aicsimageio only require a lower bound on dask, but dask
# cuts new releases very often, so leaving it unpinned makes pip's resolver
# backtrack through hundreds of dask versions. Pinning skips that search.
dask[array]==2026.7.1
# Python <3.13: tensorflow<2.17 hard-requires numpy<2.0.0, and pinning
# numpy==1.26.4 here (rather than leaving it unpinned) skips pip having to
# backtrack through scipy/opencv-python's full release histories to find a
# numpy<2.0-compatible version of each.
# Python >=3.13: numpy<2.1 and tensorflow<2.20 ship no Python 3.13 wheels at
# all, so both floors move up together instead.
numpy==1.26.4; python_version < "3.13"
numpy>=2.1; python_version >= "3.13"
scipy==1.14.1
scikit-image
matplotlib
Pillow

# Notebook environment
jupyter

# Image I/O
opencv-python==4.10.0.84
# Pinned: aicsimageio 4.12.1/4.13.0 cap dask at <=2023.5.0, which conflicts
# with the dask pin above and sent pip walking back through aicsimageio's
# whole history (down to versions that predate the 'nd2' extra) looking for
# an alternative. 4.14.0 has no dask ceiling and includes the nd2 extra.
aicsimageio[nd2]==4.14.0
nd2reader
# aicsimageio==4.14.0 unconditionally requires lxml<5, but lxml<5 has no
# Python 3.13 wheels and fails to build from source on Windows without a
# full MSVC toolchain. ome-types -- aicsimageio's own lxml consumer --
# declares lxml>=5.3 as correct for Python 3.13, so raising the floor here
# is safe; it's aicsimageio's cap that's stale, not a real incompatibility.
# This line must ALSO be listed in overrides.txt (see the command at the top
# of this file) so it takes precedence over that cap on 3.13. overrides.txt
# is kept separate from this file -- and deliberately doesn't repeat the
# extras-bearing lines above (napari[all], aicsimageio[nd2], dask[array]) --
# because an override entry with extras isn't guaranteed to propagate them
# the same way a normal requirement does.
lxml<5,>=4.6; python_version < "3.13"
lxml>=5.3; python_version >= "3.13"

# Segmentation models
tensorflow>=2.16,<2.17; python_version < "3.13"
tensorflow>=2.20; python_version >= "3.13"
csbdeep
stardist
cellpose>=3.0
