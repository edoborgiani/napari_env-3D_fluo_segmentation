# uv/pip override file for the Python 3.13 dependency set (requirements-py313.txt).
#
# aicsimageio==4.14.0 unconditionally pins lxml<5,>=4.6. lxml<5 has no
# Python 3.13 wheels on any platform, so pip/uv falls back to building
# lxml==4.9.4 from source -- which then fails on Windows unless the full
# MSVC C++ Build Tools (and prebuilt libxml2/libxslt) are installed.
#
# ome-types (the aicsimageio dependency that actually calls into lxml)
# explicitly declares lxml>=5.3 as the correct version for Python 3.13, so
# aicsimageio's <5 cap is just stale packaging metadata, not a real
# incompatibility. This override tells the resolver to ignore that cap and
# use a modern lxml with prebuilt wheels instead.
#
# Usage (uv only -- plain pip has no equivalent override mechanism, only
# --constraint, which can narrow but not widen another package's bound):
#   uv pip install -r requirements-py313.txt --override overrides-py313.txt
lxml>=5.3
